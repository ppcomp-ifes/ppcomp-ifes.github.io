Versão com shell.nix do ambiente Python com uv, CUDA e dependências nativas. Salve na raiz do projeto e entre com nix-shell.
Para fazer a instalação do torch corretamente com acesso à GPU, utilize o comando abaixo:
uv add torch torchvision torchaudio --index https://download.pytorch.org/whl/cu132 

{ pkgs ? import <nixpkgs> {
  config.allowUnfree = true;
} }:

pkgs.mkShell {
  packages = with pkgs; [
    # Python
    python313
    uv

    # CUDA
    cudaPackages.cudatoolkit

    # C/C++
    gcc
    pkg-config

    # Bibliotecas nativas
    libxcb
    libX11
    libGL
    glib
  ];

  shellHook = ''
    # Exports necessários para funcionamento de bibliotecas externas com CUDA
    export CUDA_PATH="${pkgs.cudaPackages.cudatoolkit}"
    export PATH="$CUDA_PATH/bin:$PATH"

    export LD_LIBRARY_PATH="${
      pkgs.lib.makeLibraryPath [
        pkgs.stdenv.cc.cc.lib
        pkgs.cudaPackages.cudatoolkit
        pkgs.libxcb
        pkgs.libX11
        pkgs.libGL
        pkgs.glib
      ]
    }:/run/opengl-driver/lib:$LD_LIBRARY_PATH"
  '';
}
